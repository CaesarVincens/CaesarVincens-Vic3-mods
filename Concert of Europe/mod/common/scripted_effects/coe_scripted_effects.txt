
coe_build_sphere = {
	$SCOPE$ ?= {
		clear_variable_list = coe_inner_sphere
		clear_variable_list = coe_outer_sphere
		set_variable = {
			name = coe_inner_sphere_size
			value = 0
		}
		set_variable = {
			name = coe_outer_sphere_size
			value = 0
		}
	}
	every_country = {
		limit = {
			exists = $SCOPE$
			OR = {
				is_direct_subject_of = $SCOPE$
				trigger_if = {
					limit = { $SCOPE$ = { is_power_bloc_leader = yes } }
					is_in_same_power_bloc = $SCOPE$
				}
			}
			NOR = {
				$SCOPE$ = this
				is_involved_in_journal_entry = $journal$
				is_indirect_subject_of = $SCOPE$
			}
		}
		$SCOPE$ = {
			add_to_variable_list = {
				name = coe_inner_sphere
				target = prev
			}
			change_variable = {
				name = coe_inner_sphere_size
				add = 1
			}
		}
		coe_calc_sphere_influence = { SCOPE = $SCOPE$ }
	}
	every_country = {
		limit = {
			exists = $SCOPE$
			OR = {
				is_indirect_subject_of = $SCOPE$
				is_adjacent_to_country = $SCOPE$
			}
			NOR = {
				$SCOPE$ = {
					is_target_in_variable_list = {
						name = coe_inner_sphere
						target = prev
					}
				}
				$SCOPE$ = this
				is_involved_in_journal_entry = $journal$
				is_direct_subject_of = $SCOPE$
			}
		}
		$SCOPE$ = {
			add_to_variable_list = {
				name = coe_outer_sphere
				target = prev
			}
			change_variable = {
				name = coe_outer_sphere_size
				add = 1
			}
		}
		coe_calc_sphere_influence = { SCOPE = $SCOPE$ }
	}
	$SCOPE$ = {
		sort_variable_list = {
			name = coe_inner_sphere
			order = country_rank
		}
		sort_variable_list = {
			name = coe_outer_sphere
			order = country_rank
		}
	}
}

coe_save_ordered_rulers = {
	coe_save_ordered_ruler = { num = 0 }
	coe_save_ordered_ruler = { num = 1 }
	coe_save_ordered_ruler = { num = 2 }
	coe_save_ordered_ruler = { num = 3 }
	coe_save_ordered_ruler = { num = 4 }
}

coe_save_ordered_ruler = {
	scope:journal_entry = {
		ordered_scope_je_involved = {
			order_by = country_rank
			position = $num$
			ruler ?= {
				scope:journal_entry = {
					set_variable = {
						name = coe_ruler_$num$
						value = prev
					}
				}
			}
		}
	}
}

coe_calc_sphere_influence = {
	$SCOPE$ ?= {
		save_scope_as = influencing_gp
	}
	save_scope_as = influenced_country
	coe_update_variable_map_value = {
		name = coe_sphere_influence
		key = scope:influencing_gp
		value = coe_great_power_influence
	}
}

coe_update_variable_map_value = {
	if = {
		limit = {
			is_key_in_variable_map = {
				name = $name$
				target = $key$
			}
		}
		remove_from_variable_map = {
			name = $name$
			key = $key$
		}
	}
	add_to_variable_map = {
		name = $name$
		key = $key$
		value = $value$
	}
}
