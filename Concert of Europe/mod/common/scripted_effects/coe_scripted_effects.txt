
coe_build_sphere = {
	$SCOPE$ ?= {
		clear_variable_list = coe_inner_sphere
		clear_variable_list = coe_outer_sphere
		set_variable = {
			name = coe_inner_sphere_size
			value = 0
		}
		set_variable = {
			name = coe_outer_sphere_size
			value = 0
		}
	}
	every_country = {
		limit = {
			exists = $SCOPE$
			OR = {
				is_direct_subject_of = $SCOPE$
				trigger_if = {
					limit = { $SCOPE$ = { is_power_bloc_leader = yes } }
					is_in_same_power_bloc = $SCOPE$
				}
			}
			NOR = {
				$SCOPE$ = this
				is_involved_in_journal_entry = je_coe_concert_of_europe
				is_indirect_subject_of = $SCOPE$
			}
		}
		$SCOPE$ = {
			add_to_variable_list = {
				name = coe_inner_sphere
				target = prev
			}
			change_variable = {
				name = coe_inner_sphere_size
				add = 1
			}
		}
	}
	every_country = {
		limit = {
			exists = $SCOPE$
			OR = {
				is_indirect_subject_of = $SCOPE$
				any_neighbouring_state = {
					owner ?= $SCOPE$
				}
			}
			NOR = {
				$SCOPE$ = {
					is_target_in_variable_list = {
						name = coe_inner_sphere
						target = prev
					}
				}
				$SCOPE$ = this
				is_involved_in_journal_entry = je_coe_concert_of_europe
				is_direct_subject_of = $SCOPE$
			}
		}
		$SCOPE$ = {
			add_to_variable_list = {
				name = coe_outer_sphere
				target = prev
			}
			change_variable = {
				name = coe_outer_sphere_size
				add = 1
			}
		}
	}
	$SCOPE$ = {
		sort_variable_list = {
			name = coe_inner_sphere
			order = country_rank
		}
		sort_variable_list = {
			name = coe_outer_sphere
			order = country_rank
		}
	}
}

coe_save_ordered_rulers = {
	coe_save_ordered_ruler = { num = 0 }
	coe_save_ordered_ruler = { num = 1 }
	coe_save_ordered_ruler = { num = 2 }
	coe_save_ordered_ruler = { num = 3 }
	coe_save_ordered_ruler = { num = 4 }
}
coe_save_ordered_ruler = {
	global_var:je_coe_concert_of_europe_var = {
		ordered_scope_je_involved = {
			order_by = country_rank
			position = $num$
			ruler ?= {
				global_var:je_coe_concert_of_europe_var = {
					set_variable = {
						name = coe_ruler_$num$
						value = prev
					}
				}
			}
		}
	}
}
